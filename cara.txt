OnTokenValidated = context =>
            {
                System.Text.StringBuilder sb = new();
                sb.AppendLine($"===== PING CLAIMS DUMP {DateTime.Now:O} =====");
 
                if (context.Principal is not null)
                {
                    foreach (ClaimsIdentity identity in context.Principal.Identities)
                    {
                        sb.AppendLine($"--- Identity: {identity.AuthenticationType} ---");
 
                        foreach (Claim claim in identity.Claims)
                        {
                            sb.AppendLine($"  {claim.Type} = {claim.Value}  [{claim.ValueType}]");
                        }
                    }
                }
                else
                {
                    sb.AppendLine("context.Principal was NULL");
                }
 
                System.IO.File.AppendAllText(@"C:\temp\ping-claims-dump.txt", sb.ToString() + Environment.NewLine);
 
                MapPingRoleClaims(context.Principal, ping);
                MapPingProfileClaims(context.Principal, ping);
                return Task.CompletedTask;
            },
        };
